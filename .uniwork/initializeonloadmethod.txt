 Description
 Allow an editor class method to be initialized when Unity loads without action from the user.
using UnityEngine;
using UnityEditor;
class MyClass
{
 [InitializeOnLoadMethod]
 static void OnProjectLoadedInEditor()
 {
 Debug.Log("Project loaded in Unity Editor");
 }
}
